# Weapon Attacks & Spell-Casting Magic Items — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give Brunella's Enspelled Longbow +2 a home — an Attacks panel on the Dashboard, and an Encounter item that spends a charge to cast Ensnaring Strike.

**Architecture:** A new `Weapon` type authored in the character JSON feeds a pure `src/lib/weapons.ts` and a new `AttacksPanel` beside the AC panel. Two additive optional fields on the existing `Resource` (`rechargeDice`, `itemSpell`) turn a plain counter into an item that casts a spell it carries in `innateSpells` with `source: "item"`. No new store actions: the existing `useResource` / `refundResource` do the work.

**Tech Stack:** React 18, TypeScript, Zustand (`persist`), Tailwind with M3-ish tokens, Vitest (+ jsdom per-file for component tests), Vite.

**Spec:** `docs/superpowers/specs/2026-08-02-magic-bow-design.md` — read it first; it carries the adversarial reviewer's fixes and explains *why* several of these steps are shaped the way they are.

## Global Constraints

- **Never commit on `main`.** Work happens on `feat/magic-bow-attacks`. Never push without asking.
- **UI copy is English.** The Dashboard and Encounter pages are English; new strings match the page they land on.
- **Vitest globals are OFF.** Every test file imports `describe/it/expect` explicitly from `vitest`. Component tests declare `// @vitest-environment jsdom` on line 1 and call `afterEach(cleanup)` themselves — RTL auto-cleanup does not run.
- **`c.weapons` is `undefined`, not `[]`,** for `sampleWizard`, XML imports and every currently-persisted store. Every read site uses `c.weapons ?? []`.
- **No persist version bump.** `weapons` is additive and optional with a read-side default; a migration would have no payload. Do not add one.
- **Resources are keyed by `name`** in every store action. The bow's resource is `"Longbow +2 — Charges"` (em-dash) and the weapon is `"Longbow +2"`. Keep them distinct.
- **One Bash command per call; never chain with `&&`.**
- Run after every task: `npx tsc --noEmit`, then `npx vitest run`.

import { abilityMod, abilityScore } from "@/lib/abilities";
import type { Character, Weapon } from "@/types/character";

/**
 * Weapon attack and damage arithmetic, kept pure and separate from the panel so
 * it can be tested without a DOM — the same split `armor.ts` uses for AC.
 */

/** Signed display for a modifier: 3 -> "+3", -1 -> "-1". */
export function fmtBonus(n: number): string {
  return n >= 0 ? `+${n}` : `${n}`;
}

/** Ability modifier driving this weapon, feat and magic score bonuses included. */
function weaponAbilityMod(c: Character, w: Weapon): number {
  return abilityMod(abilityScore(c, w.ability));
}

/** To-hit: ability + proficiency (only if proficient) + the weapon's magic bonus. */
export function weaponAttackBonus(c: Character, w: Weapon): number {
  return (
    weaponAbilityMod(c, w) +
    (w.proficient ? c.proficiencyBonus : 0) +
    (w.magicBonus ?? 0)
  );
}

/** Flat damage added to the dice: ability + magic bonus. Proficiency never applies. */
export function weaponDamageBonus(c: Character, w: Weapon): number {
  return weaponAbilityMod(c, w) + (w.magicBonus ?? 0);
}

/**
 * "1d8+5 Piercing". A zero bonus is omitted rather than rendered as "+0", which
 * reads like a bug on the sheet.
 */
export function weaponDamageLabel(c: Character, w: Weapon): string {
  const bonus = weaponDamageBonus(c, w);
  const bonusText = bonus === 0 ? "" : fmtBonus(bonus);
  return `${w.damageDice}${bonusText} ${w.damageType}`;
}
```

/**
 * One "label … signed value" line in a stat breakdown. Shared by the AC and
 * Attacks panels so the two cannot drift apart visually.
 */
export default function BreakdownRow({
  label,
  value,
  hint,
  signed,
}: {
  label: string;
  value: number | string;
  hint?: string;
  signed?: boolean;
}) {
  const shown =
    typeof value === "number" && signed
      ? value >= 0
        ? `+${value}`
        : `${value}`
      : value;
  return (
    <li className="flex items-center justify-between gap-2">
      <span className="text-on-surface-variant truncate">
        {label}
        {hint && <span className="text-outline text-xs"> · {hint}</span>}
      </span>
      <span className="font-mono text-on-surface shrink-0">{shown}</span>
    </li>
  );
}
```

import { useCharacter } from "@/store/character";
import { abilityMod, abilityScore } from "@/lib/abilities";
import {
  fmtBonus,
  weaponAttackBonus,
  weaponDamageLabel,
} from "@/lib/weapons";
import type { Weapon } from "@/types/character";
import BreakdownRow from "../ui/BreakdownRow";

const ABILITY_LABEL: Record<string, string> = {
  str: "Strength",
  dex: "Dexterity",
  con: "Constitution",
  int: "Intelligence",
  wis: "Wisdom",
  cha: "Charisma",
};

/**
 * Weapon attacks, mirroring the Armor Class panel. Numbers only — the app is a
 * companion for play with physical dice, so it reports modifiers and never
 * rolls. The breakdown exists so a wrong total is visible rather than merely
 * wrong.
 */
export default function AttacksPanel() {
  const c = useCharacter((s) => s.character);
  const weapons = c.weapons ?? [];
  if (weapons.length === 0) return null;

  return (
    <div className="bg-surface-container border border-outline-variant/30 rounded-xl p-md relative overflow-hidden">
      <div className="leather-noise absolute inset-0" />
      <div className="relative space-y-md">
        <h3 className="font-serif text-title-sm text-primary">Attacks</h3>
        {weapons.map((w, i) => (
          <WeaponBlock key={`${w.name}-${i}`} weapon={w} />
        ))}
      </div>
    </div>
  );
}

function WeaponBlock({ weapon }: { weapon: Weapon }) {
  const c = useCharacter((s) => s.character);
  const attack = weaponAttackBonus(c, weapon);
  const abilityBonus = abilityMod(abilityScore(c, weapon.ability));

  return (
    <div>
      <div className="flex items-center gap-md">
        <div className="shrink-0 w-20 h-20 rounded-full border-2 border-primary/60 flex flex-col items-center justify-center bg-surface-container-low">
          <span className="font-serif text-3xl text-primary leading-none">
            {fmtBonus(attack)}
          </span>
          <span className="label-caps text-outline text-[9px] mt-0.5">ATK</span>
        </div>

        <div className="flex-1 min-w-0">
          <p className="font-serif text-on-surface truncate">{weapon.name}</p>
          <p className="font-mono text-sm text-on-surface-variant">
            {weaponDamageLabel(c, weapon)}
          </p>
          {weapon.range && (
            <p className="text-xs text-outline">{weapon.range}</p>
          )}
          {weapon.properties && weapon.properties.length > 0 && (
            <p className="text-xs text-outline">
              {weapon.properties.join(" · ")}
            </p>
          )}
        </div>
      </div>

      <ul className="space-y-0.5 text-sm mt-sm">
        <BreakdownRow
          label={ABILITY_LABEL[weapon.ability] ?? weapon.ability}
          value={abilityBonus}
          signed
        />
        {/* Rendered even when it does not apply: a missing row is
            indistinguishable from a bug. */}
        <BreakdownRow
          label="Proficiency"
          value={weapon.proficient ? fmtBonus(c.proficiencyBonus) : "—"}
          hint={weapon.proficient ? undefined : "not proficient"}
        />
        {weapon.magicBonus ? (
          <BreakdownRow label="Magic bonus" value={weapon.magicBonus} signed />
        ) : null}
      </ul>

      {weapon.note && (
        <p className="text-[11px] text-on-surface-variant italic mt-1 leading-snug">
          {weapon.note}
        </p>
      )}
    </div>
  );
}
```

/**
 * Legal range for a dice-recharge entry. "2d6" is 2..12 — sizing it from the
 * right-hand side alone would wrongly allow 1 and cap at 6.
 */
export function parseRechargeDice(
  dice: string,
): { min: number; max: number } | null {
  const m = /^(\d+)d(\d+)$/.exec(dice.trim());
  if (!m) return null;
  const count = Number(m[1]);
  const sides = Number(m[2]);
  if (count < 1 || sides < 1) return null;
  return { min: count, max: count * sides };
}
```

```tsx
import { useState } from "react";
import { useCharacter } from "@/store/character";
import { parseRechargeDice } from "@/lib/rechargeDice";
import type { Resource } from "@/types/character";
import Icon from "./ui/Icon";

/**
 * "Regains 1d6 charges daily at dawn" — you roll the physical die, the app
 * takes the number. Dawn is a clock event, not a rest, so this lives on the
 * item rather than inside the Rest menu.
 */
export default function DawnRecharge({ resource }: { resource: Resource }) {
  const refund = useCharacter((s) => s.refundResource);
  const [open, setOpen] = useState(false);
  const [value, setValue] = useState("");

  if (!resource.rechargeDice) return null;
  const range = parseRechargeDice(resource.rechargeDice);
  const n = Number(value);
  const valid =
    value.trim() !== "" &&
    Number.isInteger(n) &&
    n >= 1 &&
    (range === null || (n >= range.min && n <= range.max));
  const gain = valid ? Math.min(n, resource.used) : 0;

  function commit() {
    if (!valid) return;
    refund(resource.name, n);
    setValue("");
    setOpen(false);
  }

  return (
    <>
      <button
        onClick={() => setOpen((o) => !o)}
        disabled={resource.used <= 0}
        className="shrink-0 btn-icon !w-7 !h-7 disabled:opacity-30"
        aria-label={`Dawn recharge — roll ${resource.rechargeDice}`}
        title={
          resource.used > 0
            ? `Dawn — roll ${resource.rechargeDice}`
            : "Already at full charges"
        }
      >
        <Icon name="wb_sunny" size={14} />
      </button>

      {open && (
        <div className="w-full mt-1 flex items-center gap-2 border-t border-outline-variant/30 pt-2 animate-fade-in">
          <span className="text-[11px] text-on-surface-variant">
            Roll {resource.rechargeDice} and enter the result.
          </span>
          <input
            type="number"
            inputMode="numeric"
            placeholder={resource.rechargeDice}
            min={range?.min ?? 1}
            max={range?.max}
            value={value}
            onChange={(e) => setValue(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === "Enter") commit();
            }}
            className="input-inset w-16 text-center font-mono"
            aria-label={`Result of ${resource.rechargeDice}`}
            autoFocus
          />
          <button onClick={commit} disabled={!valid} className="btn-brass text-sm disabled:opacity-40">
            Recharge
            {gain > 0 && <span className="ml-1 text-on-primary/80">+{gain}</span>}
          </button>
        </div>
      )}
    </>
  );
}
```
